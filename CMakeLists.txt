cmake_minimum_required(VERSION 3.22)
project(FamiliarPaint VERSION 0.0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Concurrent Network)
add_library(paint-core src/document.cpp src/document.h src/drawing.cpp src/drawing.h src/commands.cpp src/commands.h)
target_link_libraries(paint-core PUBLIC Qt6::Gui)
target_include_directories(paint-core PUBLIC src)
add_library(paint-ui src/canvas.cpp src/canvas.h src/window.cpp src/window.h src/theme.cpp src/theme.h src/agent.cpp src/agent.h)
target_link_libraries(paint-ui PUBLIC paint-core Qt6::Widgets Qt6::Concurrent Qt6::Network)
add_executable(familiar-paint src/main.cpp)
target_link_libraries(familiar-paint PRIVATE paint-ui)
include(GNUInstallDirs)
install(TARGETS familiar-paint RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/io.github.tcballard.FamiliarPaint.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/io.github.tcballard.FamiliarPaint.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(paint-tests tests/native/paint_tests.cpp)
  target_link_libraries(paint-tests PRIVATE paint-ui Qt6::Test)
  add_executable(paint-preview tests/native/preview.cpp)
  target_link_libraries(paint-preview PRIVATE paint-ui Qt6::Test)
  add_test(NAME paint-tests COMMAND paint-tests)
  add_test(NAME paint-preview-compact COMMAND paint-preview preview-compact.png 800 600 verify)
  add_test(NAME paint-preview-hidpi COMMAND paint-preview preview-hidpi.png 1000 720 verify)
  set_tests_properties(paint-tests paint-preview-compact PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  set_tests_properties(paint-preview-hidpi PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=1.5")
endif()
